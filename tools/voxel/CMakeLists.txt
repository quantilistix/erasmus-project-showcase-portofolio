cmake_minimum_required(VERSION 3.20)
project(VoxelShowcase LANGUAGES C CXX)

get_filename_component(VOXEL_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../hand-coded/voxel-based-game" ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 CONFIG REQUIRED)
find_package(TBB CONFIG REQUIRED COMPONENTS tbb)
find_package(Threads REQUIRED)

# Stage headers without the machine-specific, checked-in pch.hpp.gch.
# configure_file also makes CMake reconfigure when an original header changes.
file(GLOB headers CONFIGURE_DEPENDS "${VOXEL_SOURCE_DIR}/include/*.hpp" "${VOXEL_SOURCE_DIR}/include/*.h")
foreach(header IN LISTS headers)
    get_filename_component(name "${header}" NAME)
    configure_file("${header}" "${CMAKE_CURRENT_BINARY_DIR}/headers/${name}" COPYONLY)
endforeach()
file(GLOB sources CONFIGURE_DEPENDS "${VOXEL_SOURCE_DIR}/src/*.cpp")
add_executable(Application "${VOXEL_SOURCE_DIR}/main.cpp" ${sources} "${VOXEL_SOURCE_DIR}/dependencies/src/glad.c")
target_include_directories(Application PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/headers" "${VOXEL_SOURCE_DIR}/dependencies/include")
target_compile_definitions(Application PRIVATE GLFW_INCLUDE_NONE NOMINMAX)
target_link_libraries(Application PRIVATE OpenGL::GL glfw TBB::tbb Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(Application PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(Application PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_BINARY_DIR}/bin")
endforeach()
if(MSVC)
    target_compile_options(Application PRIVATE /W3 /permissive-)
else()
    target_compile_options(Application PRIVATE -Wall -Wextra)
endif()
set_property(TARGET Application PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${VOXEL_SOURCE_DIR}")
add_custom_target(run COMMAND $<TARGET_FILE:Application>
    WORKING_DIRECTORY "${VOXEL_SOURCE_DIR}" DEPENDS Application USES_TERMINAL)
